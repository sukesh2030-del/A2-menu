let cart = {};

function renderMenu(filterKeyword = "") {
const catNav = document.getElementById('categoryNav');
const menuContainer = document.getElementById('menuContainer');
const navWrapper = document.getElementById('navWrapper');

catNav.innerHTML = '';
menuContainer.innerHTML = '';

const keyword = filterKeyword.toLowerCase().trim();
let totalRendered = 0;

if (keyword) {
navWrapper.style.display = "none";
} else {
navWrapper.style.display = "block";
}

menuData.forEach((cat, index) => {
const catId = `cat-${index}`;
const filteredItems = cat.items.filter(item => item.name.toLowerCase().includes(keyword));

if (filteredItems.length === 0) return;

totalRendered += filteredItems.length;

// Category Nav Pill
if (!keyword) {
const btn = document.createElement('button');
btn.className = `cat-btn ${index === 0 ? 'active' : ''}`;
btn.id = `pill-${catId}`;
btn.innerText = cat.category;
btn.onclick = () => scrollToCategory(catId);
catNav.appendChild(btn);
}

// Section Header
const title = document.createElement('div');
title.className = 'section-title';
title.id = catId;
title.innerText = cat.category;
menuContainer.appendChild(title);

// Items list
filteredItems.forEach(item => {
const card = document.createElement('div');
card.className = 'item-card';
card.innerHTML = `
<div class="item-details">
<div class="veg-symbol"><span></span></div>
<div class="item-name">${item.name}</div>
<div class="item-price">₹${Math.round(item.price)}</div>
</div>
<div id="btn-box-${item.id}">
${renderButton(item.id)}
</div>
`;
menuContainer.appendChild(card);
});
});

if (totalRendered === 0) {
menuContainer.innerHTML = `
<div class="no-results">
<i class="fas fa-search" style="font-size: 2rem; margin-bottom: 8px; opacity: 0.4;"></i>
<p>No dishes found matching "${filterKeyword}".</p>
</div>
`;
}
}

function renderButton(itemId) {
const qty = cart[itemId] || 0;
if (qty > 0) {
return `
<div class="add-btn-container">
<button type="button" class="qty-btn" onclick="updateQty(${itemId}, -1)">−</button>
<span class="qty-count">${qty}</span>
<button type="button" class="qty-btn" onclick="updateQty(${itemId}, 1)">+</button>
</div>
`;
}
return `<button type="button" class="btn-add" onclick="updateQty(${itemId}, 1)">ADD</button>`;
}

function updateQty(itemId, change) {
const currentQty = cart[itemId] || 0;
const newQty = currentQty + change;
if (newQty <= 0) {
delete cart[itemId];
} else {
cart[itemId] = newQty;
}

const box = document.getElementById(`btn-box-${itemId}`);
if (box) box.innerHTML = renderButton(itemId);

updateCartBar();
}

function updateCartBar() {
const cartBar = document.getElementById('cartBar');
let totalItems = 0;
let totalPrice = 0;

for (let id in cart) {
const qty = cart[id];
if (qty > 0 && itemLookup[id]) {
totalItems += qty;
totalPrice += Math.round(itemLookup[id].price) * qty;
}
}

if (totalItems > 0) {
cartBar.style.display = 'flex';
document.getElementById('cartCount').innerText = `${totalItems} ${totalItems === 1 ? 'ITEM' : 'ITEMS'}`;
document.getElementById('cartTotal').innerText = `Total: ₹${totalPrice}`;
} else {
cartBar.style.display = 'none';
}
}

function scrollToCategory(catId) {
const element = document.getElementById(catId);
if (!element) return;
const offset = 120;
const top = element.getBoundingClientRect().top + window.pageYOffset - offset;
window.scrollTo({ top, behavior: 'smooth' });
}

function filterMenu() {
const query = document.getElementById('searchInput').value;
renderMenu(query);
}

// Scrollspy to auto-highlight category tabs
window.addEventListener('scroll', () => {
const searchVal = document.getElementById('searchInput').value.trim();
if (searchVal) return;

const sections = document.querySelectorAll('.section-title');
let current = "";

sections.forEach(sec => {
const secTop = sec.getBoundingClientRect().top;
if (secTop <= 140) {
current = sec.getAttribute('id');
}
});

if (current) {
document.querySelectorAll('.cat-btn').forEach(btn => {
btn.classList.remove('active');
if (btn.id === `pill-${current}`) {
btn.classList.add('active');
btn.scrollIntoView({ behavior: 'smooth', inline: 'nearest', block: 'nearest' });
}
});
}
});

function sendWhatsAppOrder() {
const name = document.getElementById('custName').value.trim();
const address = document.getElementById('custAddress').value.trim();
const phone = document.getElementById('custPhone').value.trim();

if (!name || !address || !phone) {
alert('Kripya saari details (Naam, Pata, Phone) bharein!');
return;
}

if (!/^\d{10}$/.test(phone)) {
alert('Kripya 10-digit valid mobile number enter karein!');
return;
}

let orderText = `*New Order - A2 Restaurant (Pure Veg)*\n`;
orderText += `--------------------------------\n`;
orderText += `👤 *Customer:* ${name}\n`;
orderText += `📞 *Phone:* ${phone}\n`;
orderText += `📍 *Address:* ${address}\n`;
orderText += `--------------------------------\n`;
orderText += `*Ordered Items:*\n`;

let grandTotal = 0;
for (let id in cart) {
const qty = cart[id];
if (qty > 0 && itemLookup[id]) {
const item = itemLookup[id];
const itemTotal = Math.round(item.price) * qty;
grandTotal += itemTotal;
orderText += `• ${item.name} × ${qty} = ₹${itemTotal}\n`;
}
}

orderText += `--------------------------------\n`;
orderText += `*Grand Total: ₹${grandTotal}*\n`;
orderText += `*Delivery Fee: ₹0 (Free)*\n\n`;
orderText += `_Order sent via QR Online Menu_`;

const encodedMessage = encodeURIComponent(orderText);
const whatsappUrl = `https://wa.me/${RESTAURANT_WHATSAPP_NUMBER}?text=${encodedMessage}`;

window.open(whatsappUrl, '_blank');
closeOrderModal();
}
